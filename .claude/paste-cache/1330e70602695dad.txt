            console.log("createTable");
        
            var allegati = JSON.parse("[]");


            if (!allegati || allegati.length === 0) {
                k("#riepilogo-allegati").html("<p>Nessun allegato disponibile.</p>");
                return;
            }
        
            var header = '<thead class="tableAllegati"><tr>';
            header += '<th style="word-wrap: break-word; vertical-align: top;">Nome file</th>';
            header += '<th style="word-wrap: break-word; vertical-align: top;">Impronta</th>';
            header += '<th style="word-wrap: break-word; vertical-align: top;">Firmato digitalmente</th>';
            header += '<th style="word-wrap: break-word; vertical-align: top;">Soggetto firmatario</th>';
            header += '</tr></thead>';
        
            var body = "<tbody class='tableAllegati'>";
            for (var i = 0; i < allegati.length; i++) {
                var a = allegati[i];
                var nome = a.nomeFile;
                var impronta = a.impronta;
                var firmato = a.firmato === true ? "Sì" : "No";
                var identificativo = "";
                if (a.identificativoFirmatario && a.identificativoFirmatario != null) {
                    identificativo = a.identificativoFirmatario;
                }
        
                body += "<tr>";
                body += "<td style='word-wrap: break-word'>" + nome + "</td>";
                body += "<td style='word-wrap: break-word'>" + impronta + "</td>";
                body += "<td style='word-wrap: break-word'>" + firmato + "</td>";
                body += "<td style='word-wrap: break-word'>" + identificativo + "</td>";
                body += "</tr>";
            }
            body += "</tbody>";
        
            // Inserisce tabella completa nel contenitore
            var htmlTable = header + body;
            k("#riepilogo-allegati").html(htmlTable);
        
            // Wrappa la tabella dentro <table> solo dopo averla inserita nel DOM
            k('.tableAllegati').wrapAll("<table class='table table-bordered' style='table-layout: fixed; width: 100%;'></table>");
        }
