listaTotale = scrivaniaCittadinoMiddlewareService.getListaComunicazioni(comunicazioneFilters, loggedUser.getUserId(), cur, ScrivaniaCittadinoPortletKeys.DEFAULT_DELTA, themeDisplay.getSiteDefaultLocale(), true);
            
            List<ComunicazioneDto> paginaSuccessiva = scrivaniaCittadinoMiddlewareService.getListaComunicazioni(comunicazioneFilters, loggedUser.getUserId(), cur + 1, ScrivaniaCittadinoPortletKeys.DEFAULT_DELTA, themeDisplay.getSiteDefaultLocale(), true);
            
            if (Validator.isNotNull(paginaSuccessiva) && !paginaSuccessiva.isEmpty()) {
                hasNext = true;
            }
        }
        catch (Exception e) {
            _log.error("doServeResource() :: " + e.getMessage(), e);
            auditService.saveErrorLog(resourceRequest, e.getMessage());
        }
        responseMap.put("listaComunicazioni", listaTotale);
        responseMap.put("hasNext", hasNext);
        responseMap.put("cur", cur);
        String jsonObject = JSONFactoryUtil.looseSerializeDeep(responseMap);
        resourceResponse.getWriter().write(jsonObject);
